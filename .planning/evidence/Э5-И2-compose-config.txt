    command:
      - python3
      - -c
      - |
        import glob
        import os
        import sqlite3
        import sys
        import time
        from datetime import datetime, timezone

        DAY = 86400
        RETENTION_DAYS = 14

        while True:
            stamp = datetime.now(timezone.utc).strftime("%Y%m%d-%H%M%S")
            dest = f"/backups/scout-{stamp}.db"
            try:
                src = sqlite3.connect("file:/data/scout.db?mode=ro", uri=True)
                dst = sqlite3.connect(dest)
                with dst:
                    src.backup(dst)
                dst.close()
                src.close()
            except Exception as exc:  # отказ виден, цикл не прерывается
                print(f"копия базы не выполнена: {exc}", file=sys.stderr)

            cutoff = time.time() - RETENTION_DAYS * DAY
            for path in glob.glob("/backups/scout-*.db"):
                try:
                    if os.path.getmtime(path) < cutoff:
                        os.remove(path)
                except OSError as exc:
                    print(f"ротация копии не выполнена: {exc}", file=sys.stderr)

            time.sleep(DAY)
